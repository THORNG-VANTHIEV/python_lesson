Python ជាអ្វី?

Python ជាភាសាសរសេរកម្មវិធីកម្រិតខ្ពស់ ដែល Guido van Rossum បង្កើតឡើងក្នុងឆ្នាំ 1991។ វាពេញនិយមខ្លាំងព្រោះកូដអានងាយ ស្ទើរតែដូចភាសាអង់គ្លេសធម្មតា។ គេប្រើ Python សម្រាប់៖

  • បង្កើតគេហទំព័រ (Django, Flask)
  • វិភាគទិន្នន័យ និងគូរក្រាហ្វ
  • បញ្ញាសិប្បនិម្មិត (AI) និង Machine Learning
  • ធ្វើការងារដដែលៗដោយស្វ័យប្រវត្តិ ដូចជាប្តូរឈ្មោះឯកសាររាប់រយ

មធ្យមភាគពី list ទទេ

សរសេរអនុគមន៍ average(numbers) ដែល return 0 ពេល list ទទេ ជំនួសឲ្យការគាំង។

មើលចម្លើយគំរូ 
def average(numbers):
    try:
        return sum(numbers) / len(numbers)
    except ZeroDivisionError:
        return 0

print(average([80, 90, 70]))   # 80.0
print(average([]))             # 0

FILE = "todo.txt"

def load_tasks():
    tasks = []
    try:
        with open(FILE, "r", encoding="utf-8") as f:
            for line in f:
                tasks.append(line.strip())
    except FileNotFoundError:
        pass                       # មិនទាន់មានឯកសារ គឺបញ្ជីទទេ
    return tasks

def add_task(task):
    with open(FILE, "a", encoding="utf-8") as f:
        f.write(task + "\n")

while True:
    print("\n1. មើលការងារ   2. បន្ថែមការងារ   3. ចាកចេញ")
    choice = input("ជ្រើសរើស: ")
    if choice == "1":
        tasks = load_tasks()
        if not tasks:
            print("មិនទាន់មានការងារទេ")
        for i, task in enumerate(tasks, start=1):
            print(f"{i}. {task}")
    elif choice == "2":
        add_task(input("ការងារថ្មី: "))
        print("បានរក្សាទុក!")
    elif choice == "3":
        break

import secrets
import string

def make_password(length=16):
    chars = string.ascii_letters + string.digits + "!@#$%"
    password = ""
    for _ in range(length):
        password += secrets.choice(chars)
    return password

FILE = "students.txt"


def load_students():
    """អានទិន្នន័យសិស្សពីឯកសារ"""
    students = {}
    try:
        with open(FILE, "r", encoding="utf-8") as f:
            for line in f:
                name, score = line.strip().split(",")
                students[name] = float(score)
    except FileNotFoundError:
        pass
    return students


def save_students(students):
    """រក្សាទុកទិន្នន័យសិស្សទាំងអស់ចូលឯកសារ"""
    with open(FILE, "w", encoding="utf-8") as f:
        for name, score in students.items():
            f.write(f"{name},{score}\n")


def get_grade(score):
    if score >= 90:
        return "A"
    elif score >= 80:
        return "B"
    elif score >= 70:
        return "C"
    elif score >= 60:
        return "D"
    elif score >= 50:
        return "E"
    return "F"


def add_student(students):
    name = input("ឈ្មោះសិស្ស: ").strip()
    if not name:
        print("ឈ្មោះមិនអាចទទេបានទេ")
        return
    try:
        score = float(input("ពិន្ទុ (0-100): "))
    except ValueError:
        print("ពិន្ទុត្រូវតែជាលេខ!")
        return
    if not 0 <= score <= 100:
        print("ពិន្ទុត្រូវនៅចន្លោះ 0 ដល់ 100")
        return
    students[name] = score
    save_students(students)
    print(f"បានបន្ថែម {name} ដោយជោគជ័យ")


def show_report(students):
    if not students:
        print("មិនទាន់មានទិន្នន័យសិស្សទេ")
        return
    print("\n----- របាយការណ៍ពិន្ទុ -----")
    for name, score in students.items():
        print(f"{name}: {score:.1f} ({get_grade(score)})")
    scores = list(students.values())
    print(f"មធ្យមភាគថ្នាក់: {sum(scores) / len(scores):.2f}")
    best = max(students, key=students.get)
    print(f"សិស្សពូកែបំផុត: {best} ({students[best]:.1f})")


def main():

        choice = input("ជ្រើសរើស: ")
        if choice == "1":
            add_student(students)
        elif choice == "2":
            show_report(students)
        elif choice == "3":

            break
        else:
            print("ជម្រើសមិនត្រឹមត្រូវ")
